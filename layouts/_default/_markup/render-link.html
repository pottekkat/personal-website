{{- $dest := .Destination }} {{- $external := strings.HasPrefix $dest "http" }} {{- /* A printed page can't follow links, so each link out of the site gets a number, and print_links.html lists the numbered addresses at the end of the post. The numbers carry on from the post's footnotes, so the two read as one series. A link whose text is already its address (as in most footnotes) needs no number, and neither does a linked image. The listed address is the one written in the post, without the tag added below. markdownify runs this hook as the home page, which has no list, so a link it renders (a cover's photo credit) stays unnumbered. */}} {{- $printRef := 0 }} {{- $text := string .Text }} {{- if and $external (not .Page.IsHome) (ne $text $dest) (not (strings.Contains $text ")* {0,3}\[\^([^\]]+)\]:` $raw }} {{- $defs = $defs | append (index . 1) }} {{- end }} {{- $shown := slice }} {{- range $defs }} {{- $label := . }} {{- $own := 0 }} {{- range $defs }}{{ if eq . $label }}{{ $own = add $own 1 }}{{ end }}{{ end }} {{- if and (gt (strings.Count (printf "[^%s]" $label) $raw) $own) (not (in $shown $label)) }} {{- $shown = $shown | append $label }} {{- end }} {{- end }} {{- .Page.Store.Set "printLinksOffset" (len $shown) }} {{- .Page.Store.Set "printLinksCounted" true }} {{- end }} {{- $printRef = add $printRef (.Page.Store.Get "printLinksOffset") }} {{- end }} {{- if and $external (ne (urls.Parse $dest).Host "web.archive.org") (not (strings.Contains $dest "utm_source=")) }} {{- /* Tag outbound links for the other site's analytics. The tag joins any query the link already has, and goes before the #fragment. An archive.org copy is left alone: the tag would change which page it looks up. A link that already has a utm_source keeps its own. */}} {{- $base := index (split $dest "#") 0 }} {{- $frag := strings.TrimPrefix $base $dest }} {{- $sep := cond (strings.Contains $base "?") "&" "?" }} {{- $dest = printf "%s%sutm_source=navendu_blog&utm_medium=referral&utm_campaign=%s%s" $base $sep (.Page.Title | urlize) $frag }} {{- end -}} {{ .Text }} {{- /* This comment removes trailing newlines. */ -}}
