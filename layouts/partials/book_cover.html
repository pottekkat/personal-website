{{- /* Renders a book cover through Hugo's image pipeline (WebP + srcset + dominant-color background + explicit dimensions). Mirrors the prod-only processing gate used by partials/cover.html. Resolves the image via resources.Get; static/ files are visible to the resource pipeline through the static->assets module mount in config.yml. If the resource is missing the partial falls back to a plain pointing at the original path so the page still renders. ctx (dict): image - path from frontmatter (e.g. /images/books/covers/foo.jpg) title - book title (used for alt text) class - class applied to the rendered sizes - sizes attribute for srcset */ -}} {{- $image := .image -}} {{- $title := .title -}} {{- $class := .class -}} {{- $sizes := .sizes | default "(min-width: 768px) 250px, 50vw" -}} {{- $cover := "" -}} {{- with $image -}} {{- $cover = resources.Get (strings.TrimPrefix "/" .) -}} {{- end -}} {{- if $cover -}} {{- $prod := (hugo.IsProduction | or (eq site.Params.env "production")) -}} {{- $processable := in (slice "jpeg" "jpg" "png" "tif" "bmp" "gif" "webp") $cover.MediaType.SubType -}} {{- if and $prod $processable -}} {{- $widths := slice 160 240 360 480 720 -}} {{- $dominant := "" -}} {{- with ($cover.Resize "60x").Colors -}}{{- $dominant = index . 0 -}}{{- end -}} {{- $fallback := $cover.Resize "360x q85" -}} {{- else -}} {{- end -}} {{- else -}} {{- end -}}
