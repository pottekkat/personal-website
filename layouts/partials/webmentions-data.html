{{- /* Fetch all webmentions for the domain in a single API call. Uses caching and incremental updates with 'since' parameter. Returns: Map of URL -> mentions array */ -}} {{- $domain := site.Params.webmentionDomain | default (urls.Parse site.BaseURL).Host -}} {{- $token := os.Getenv "WEBMENTION_TOKEN" | default site.Params.webmentionToken -}} {{- $byTarget := dict -}} {{- if not $token -}} {{- warnf "No webmention token found. Set WEBMENTION_TOKEN environment variable." -}} {{- else -}} {{- /* Try to get last fetch time from data file, fallback to 30 days ago */ -}} {{- $since := "" -}} {{- with site.Data.webmentions.last_fetch -}} {{- with .timestamp -}} {{- $since = . -}} {{- end -}} {{- else -}} {{- /* Fallback: fetch last 30 days */ -}} {{- $thirtyDaysAgo := now.AddDate 0 0 -30 -}} {{- $since = $thirtyDaysAgo.Format "2006-01-02T15:04:05-07:00" -}} {{- end -}} {{- /* Build API URL - just token, no domain parameter */ -}} {{- $apiUrl := printf "https://webmention.io/api/mentions.jf2?token=%s" $token -}} {{- if $since -}} {{- $apiUrl = printf "%s&since=%s" $apiUrl $since -}} {{- end -}} {{- /* Add per-page for pagination */ -}} {{- $apiUrl = printf "%s&per-page=100" $apiUrl -}} {{- /* Create cache key that refreshes every hour */ -}} {{- $cacheKey := printf "webmentions-%s-%s" $domain (now.Format "2006-01-02-15") -}} {{- $opts := dict "key" $cacheKey "method" "GET" -}} {{- /* Fetch webmentions with error handling */ -}} {{- $allMentions := slice -}} {{- with try (resources.GetRemote $apiUrl $opts) -}} {{- with .Err -}} {{- warnf "Failed to fetch webmentions: %s" . -}} {{- else -}} {{- with .Value -}} {{- $data := . | transform.Unmarshal -}} {{- with $data.children -}} {{- $allMentions = . -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- /* Merge with existing cached mentions from data files */ -}} {{- $existingMentions := slice -}} {{- with site.Data.webmentions.mentions -}} {{- $existingMentions = . -}} {{- end -}} {{- /* Combine and deduplicate by wm-id */ -}} {{- $mentionMap := dict -}} {{- range $existingMentions -}} {{- $wmId := index . "wm-id" -}} {{- if $wmId -}} {{- $mentionMap = merge $mentionMap (dict (string $wmId) .) -}} {{- end -}} {{- end -}} {{- range $allMentions -}} {{- $wmId := index . "wm-id" -}} {{- if $wmId -}} {{- $mentionMap = merge $mentionMap (dict (string $wmId) .) -}} {{- end -}} {{- end -}} {{- /* Group by target URL */ -}} {{- $byTarget = dict -}} {{- range $mentionMap -}} {{- $target := index . "wm-target" -}} {{- if $target -}} {{- $existing := index $byTarget $target | default slice -}} {{- $byTarget = merge $byTarget (dict $target (append $existing .)) -}} {{- end -}} {{- end -}} {{- end -}} {{- $byTarget -}}
