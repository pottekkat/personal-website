(slice .)) }} {{- if eq . site.Params.author.name }} {{- $authors = $authors | append $personRef }} {{- else }} {{- $authors = $authors | append (dict "@type" "Person" "name" .) }} {{- end }} {{- end }} {{- else }} {{- $authors = slice $personRef }} {{- end }} {{- $rfc3339 := "2006-01-02T15:04:05Z07:00" }} {{- $post := dict "@context" "https://schema.org" "@type" "BlogPosting" "description" (.Description | default .Summary | plainify | htmlUnescape | strings.TrimSpace) "wordCount" .WordCount "inLanguage" (.Language.Lang | default "en-us") "datePublished" (.PublishDate.Format $rfc3339) "dateModified" (.Lastmod.Format $rfc3339) "author" $authors "publisher" $personRef "mainEntityOfPage" (dict "@type" "WebPage" "@id" .Permalink) }} {{- if eq (len $authors) 1 }} {{- $post = merge $post (dict "author" (index $authors 0)) }} {{- end }} {{- with $page_name }} {{- $post = merge $post (dict "headline" . "name" .) }} {{- end }} {{- with .Params.keywords | default .Params.tags }} {{- $post = merge $post (dict "keywords" .) }} {{- end }} {{- /* The site's default share image doesn't show what the article is about. */}} {{- with partial "templates/share_image.html" . }} {{- if and .url (not .isDefault) }} {{- $post = merge $post (dict "image" .url) }} {{- end }} {{- end }} <script type="application/ld+json"> {{ $post | jsonify | safeJS }} </script> {{- else if and $about (eq .Permalink $about.Permalink) }} {{- /* Google's ProfilePage, for an "About Me" page on a blog. */}} {{- $profile := dict "@context" "https://schema.org" "@type" "ProfilePage" "url" .Permalink "dateModified" (.Lastmod.Format "2006-01-02T15:04:05Z07:00") "mainEntity" $person }} <script type="application/ld+json"> {{ $profile | jsonify | safeJS }} </script> {{- end }}{{/* article / profile end */}} {{- end -}}
